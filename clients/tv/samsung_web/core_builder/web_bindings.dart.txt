Future<T> providerHttpGet<T>(Uri uri,{required HttpClient client,
    required HttpOperation operation,Map<String,String> headers=const {},
    LoadToken? token,void Function()? checkCancellation,bool pauseOnAuthenticationFailure=true,
    void Function(Uri)? onResponseUri,required Future<T> Function(HttpClientResponse) readResponse}) async {
  void check() {
    checkCancellation?.call();
    if(client.closed || (token?.isCancelled??false)) throw const LoadCancelledException();
  }
  check();
  if(Zone.current[#tvGuideCacheOnly]==true)throw const LoadCancelledException();
  final callback=client.transport;
  if(callback==null) throw StateError('A physical HTTP transport is required');
  return ProviderRequestGate.instance.run(uri,() async {
    check();
    final request=jsonEncode({'url':uri.toString(),'headers':headers,'workload':_workloadName(operation.policy),
      'totalTimeoutMs':operation.remaining.inMilliseconds,
      'maximumBodyBytes':operation.policy.maximumBodyBytes,
      'maximumDecodedBytes':operation.policy.maximumDecodedBytes,
      if(operation.policy.name=='epg'&&Zone.current[#tvEpgDiskKey] is String)
        'epgDiskKey':Zone.current[#tvEpgDiskKey],
      'pauseOnAuthenticationFailure':pauseOnAuthenticationFailure,
      'idleTimeoutMs':operation.policy.idleTimeout.inMilliseconds});
    final responseText=await operation.wait((callback.callAsFunction(null,request.toJS) as JSPromise<JSString>).toDart);
    check();
    final data=objectJson(responseText.toDart);
    if(data['errorKind']=='provider-blocked') throw ProviderRequestBlocked(
      status:data['status'] as int,retryAt:DateTime.parse(data['retryAt'] as String));
    if(data['errorKind']=='cancelled') throw const LoadCancelledException();
    if(data['errorKind']=='hosted-http-blocked') throw StateError(
      'This provider uses HTTP. This TV version needs HTTPS. Ask your provider for an HTTPS address or use TVBro on Android, Windows or Samsung.');
    if(data['errorKind']=='limit') throw const HttpWorkloadException('Provider response exceeds the configured limit',isSizeLimit:true);
    if(data['errorKind']=='network') throw SocketException('Connection closed by transport');
    if(data['errorKind']!=null) throw StateError(data['error']?.toString()??'Transport capability unavailable');
    final status=data['status'] as int;
    final responseUri=Uri.parse(data['url'] as String? ?? uri.toString());
    final responseHeaders=<String,String>{if(data['headers'] is Map)
      for(final e in (data['headers'] as Map).entries)e.key.toString().toLowerCase():e.value.toString()};
    if(data['decoded']==true) {
      responseHeaders.remove('content-encoding');
      responseHeaders.remove('content-length');
    }
    final diskFile=data['bodyFile'] is Map?_DiskGuideFile.fromJson(Map<String,dynamic>.from(data['bodyFile'] as Map),callback):null;
    final bytes=data['bodyBase64'] is String ? base64Decode(data['bodyBase64'] as String)
      : Uint8List.fromList(utf8.encode(data['body'] as String? ?? ''));
    _checkLimit(bytes.length,operation.policy.maximumBodyBytes,operation.policy.name);
    for(final target in {uri,responseUri}) {
      ProviderRequestGate.instance.recordResponse(target,status,retryAfter:responseHeaders['retry-after'],
        pauseOnAuthenticationFailure:pauseOnAuthenticationFailure);
    }
    await ProviderRequestGate.instance.flushCooldowns();
    ProviderRequestGate.instance.check(uri);
    ProviderRequestGate.instance.check(responseUri);
    onResponseUri?.call(responseUri);
    return readResponse(HttpClientResponse(status,bytes,HttpHeaders(responseHeaders),diskFile));
  },timeout:operation.remaining,cancellation:token?.whenCancelled,checkCancellation:check);
}

class _WebGuideCache {
  static const maximumBytes=16*1024*1024, maximumEntries=8;
  final entries=<String,_WebGuideDownload>{};
  final fetchedAt=<String,int>{};
  final failures=<String,int>{};
  final failureReasons=<String,Map<String,Object>>{};
  int bytes=0, revision=0;
  void recordFetchedAt(String key,int at) {
    if(at<0||at>DateTime.now().millisecondsSinceEpoch)return;
    fetchedAt.remove(key);fetchedAt[key]=at;
    while(fetchedAt.length>maximumEntries)fetchedAt.remove(fetchedAt.keys.first);
  }
  Map<String,Object?> state() {
    final now=DateTime.now().millisecondsSinceEpoch,age=CanonicalTvCachePolicy._epgMaxAge.inMilliseconds;
    int? nextRefreshAt,retryAt;
    for(final at in fetchedAt.values) {
      final due=at+age;
      if(nextRefreshAt==null||due<nextRefreshAt)nextRefreshAt=due;
    }
    for(final deadline in failures.values) {
      if(deadline>now&&(retryAt==null||deadline<retryAt))retryAt=deadline;
    }
    return {'nextRefreshAt':nextRefreshAt,'retryAt':retryAt};
  }
  void remember(_WebGuideDownload value) {
    recordFetchedAt(value.key,value.at.millisecondsSinceEpoch);
    failures.remove(value.key);failureReasons.remove(value.key);revision++;
    if(value.bytes.length>maximumBytes)return;
    final previous=entries.remove(value.key);
    if(previous!=null)bytes-=previous.bytes.length;
    while(entries.isNotEmpty&&(entries.length>=maximumEntries||bytes+value.bytes.length>maximumBytes)) {
      bytes-=entries.remove(entries.keys.first)!.bytes.length;
    }
    entries[value.key]=value; bytes+=value.bytes.length;
  }
  void failed(String key,[Object? error]) {
    final now=DateTime.now().millisecondsSinceEpoch;
    final previous=failures[key];
    final deadline=error is _WebGuideDeferred?error.retryAt:
      previous!=null&&previous>now?previous:now+CanonicalTvCachePolicy.epgRetryDelay.inMilliseconds;
    final reason=error==null?(previous!=null&&previous>now?failureReasons[key]:null):_webGuideCause(error);
    if(reason?['status']==HttpStatus.forbidden) {
      failures.remove(key);failureReasons.remove(key);revision++;return;
    }
    failures.remove(key);
    while(failures.length>=maximumEntries) {
      final oldest=failures.keys.first;failures.remove(oldest);failureReasons.remove(oldest);
    }
    failures[key]=deadline;
    if(reason!=null)failureReasons[key]=reason;else failureReasons.remove(key);
    revision++;
  }
  void recordCause(Object error,Map<String,int> previous) {
    final reason=_webGuideCause(error),now=DateTime.now().millisecondsSinceEpoch;
    for(final key in failures.keys) {
      if(failures[key]!>now&&previous[key]!=failures[key]&&!failureReasons.containsKey(key))failureReasons[key]=reason;
    }
    revision++;
  }
  Object deferred(String key) {
    final deadline=failures.remove(key)!;
    failures[key]=deadline;revision++;
    // Preserve the canonical size exception for Source fallback selection.
    if(failureReasons[key]?['code']=='epg_size_limit') {
      return const HttpWorkloadException('Guide exceeds the configured limit',isSizeLimit:true);
    }
    return _WebGuideDeferred(deadline,failureReasons[key]??const {'code':'epg_unavailable'});
  }
  String export(String owner) {
    final payload=jsonEncode({'entries':[for(final value in entries.values)
      {'key':value.key,'at':value.at.millisecondsSinceEpoch,'bytes':base64Encode(value.bytes)}],
      'failures':failures,'failureReasons':failureReasons});
    return jsonEncode({'schema':1,'key':owner,'checksum':_catalogDigest(payload),'payload':payload});
  }
  bool restore(String owner,String text,{String? legacyOwner}) {
    try {
      if(text.length>24*1024*1024)return false;
      final envelope=objectJson(text), payload=envelope['payload'] as String;
      if(envelope['schema']!=1||envelope['key']!=owner&&envelope['key']!=legacyOwner||
        envelope['checksum']!=_catalogDigest(payload))return false;
      final data=objectJson(payload), now=DateTime.now().millisecondsSinceEpoch;
      final restored=<_WebGuideDownload>[]; var total=0;
      final keys=<String>{};
      for(final raw in data['entries'] as List) {
        final row=Map<String,dynamic>.from(raw as Map), key=row['key'] as String, at=row['at'] as int;
        if(!RegExp(r'^[a-f0-9]{64}$').hasMatch(key)||!keys.add(key)||at<0||at>now)return false;
        final bytes=base64Decode(row['bytes'] as String);total+=bytes.length;
        if(total>maximumBytes||restored.length>=maximumEntries)return false;
        restored.add(_WebGuideDownload(key,bytes,DateTime.fromMillisecondsSinceEpoch(at)));
      }
      final retry=<String,int>{};
      for(final entry in (data['failures'] as Map).entries) {
        final key=entry.key as String, at=entry.value as int;
        if(!RegExp(r'^[a-f0-9]{64}$').hasMatch(key)||at<0||at>now+CanonicalTvCachePolicy.epgRetryDelay.inMilliseconds||retry.length>=maximumEntries)return false;
        if(at>now)retry[key]=at;
      }
      final reasons=<String,Map<String,Object>>{};
      for(final entry in (data['failureReasons'] as Map? ?? const {}).entries) {
        final key=entry.key as String;
        final reason=_webGuideReasonFromJson(entry.value);
        if(!RegExp(r'^[a-f0-9]{64}$').hasMatch(key)||reason==null)return false;
        if(reason['status']==HttpStatus.forbidden) {retry.remove(key);continue;}
        if(retry.containsKey(key))reasons[key]=reason;
      }
      entries.clear();fetchedAt.clear();failures.clear();failureReasons.clear();bytes=0;
      for(final value in restored)remember(value);
      failures.addAll(retry);failureReasons.addAll(reasons);revision++;
      return true;
    } catch(_) { return false; }
  }
}
class EpgDownloadCache {
  const EpgDownloadCache();
  Future<EpgDownloadReceipt> fetch({required String sourceId,required Uri uri,
      required HttpClient client,required File destination,LoadToken? token,
      Map<String,String> headers=const {}}) async {
    if(token?.isCancelled??false) throw const LoadCancelledException();
    final cache=Zone.current[#tvGuideCache] as _WebGuideCache?;
    final cacheKey=_catalogDigest(jsonEncode(_catalogOrdered([sourceId,uri.toString(),headers])));
    if(Zone.current[#tvEpgDisk]==true) {
      final scope=Zone.current[#tvEpgScope] as String?;
      if(scope==null||scope.isEmpty)throw StateError('Guide owner scope is required');
      return _fetchDisk(sourceId,uri,client,destination,token,headers,cache,
        _catalogDigest(jsonEncode([scope,cacheKey])));
    }
    final cached=cache?.entries[cacheKey], now=DateTime.now();
    final age=cached==null?null:now.difference(cached.at);
    final deferred=(cache?.failures[cacheKey]??0)>now.millisecondsSinceEpoch;
    if(cached!=null&&age!=null&&!age.isNegative&&(age<CanonicalTvCachePolicy._epgMaxAge||deferred||Zone.current[#tvGuideCacheOnly]==true)) {
      cache?.recordFetchedAt(cacheKey,cached.at.millisecondsSinceEpoch);
      _memoryFiles[destination.path]=cached.bytes;
      return EpgDownloadReceipt();
    }
    if(Zone.current[#tvGuideCacheOnly]==true)throw const LoadCancelledException();
    if(deferred)throw cache!.deferred(cacheKey);
    final operation=HttpOperation(kEpgWorkload.copyWith(
      maximumBodyBytes:_webEpgMaximumBytes,maximumDecodedBytes:_webEpgMaximumBytes));
    try { await providerHttpGet<void>(uri,client:client,operation:operation,headers:headers,token:token,
      readResponse:(response) async {
        if(response.statusCode!=200) throw StateError('HTTP ${response.statusCode} downloading guide');
        await operation.readToFile(response,destination);
      }); } catch(error) {
      if(token?.isCancelled??false)rethrow;
      cache?.failed(cacheKey,error);
      if(cached==null)rethrow;
      _memoryFiles[destination.path]=cached.bytes;
      return EpgDownloadReceipt();
    }
    final bytes=_memoryFiles[destination.path]!;
    // A successful HTTP response can still contain malformed XML. Keep a
    // deferral until the parser commits, so changing guide windows cannot
    // repeatedly download an invalid feed.
    cache?.failed(cacheKey);
    return EpgDownloadReceipt(() async {
      if(token?.isCancelled??false) throw const LoadCancelledException();
      cache?.remember(_WebGuideDownload(cacheKey,bytes,DateTime.now()));
    });
  }
  Future<EpgDownloadReceipt> _fetchDisk(String sourceId,Uri uri,HttpClient client,File destination,
      LoadToken? token,Map<String,String> headers,_WebGuideCache? cache,String key) async {
    final transport=client.transport;
    if(transport==null)throw StateError('Native guide transport unavailable');
    final saved=await _guideFileRpc(transport,{'action':'lookup','key':key});
    final previous=saved is Map?_DiskGuideFile.fromJson(Map<String,dynamic>.from(saved),transport,committed:true):null;
    if(previous!=null)cache?.recordFetchedAt(key,previous.at);
    final now=DateTime.now().millisecondsSinceEpoch,deadline=cache?.failures[key]??0;
    final fresh=previous!=null&&now>=previous.at&&now-previous.at<CanonicalTvCachePolicy._epgMaxAge.inMilliseconds;
    if(previous!=null&&(fresh||deadline>now||Zone.current[#tvGuideCacheOnly]==true)) {
      _diskGuideFiles[destination.path]=previous;return EpgDownloadReceipt();
    }
    if(Zone.current[#tvGuideCacheOnly]==true)throw const LoadCancelledException();
    if(deadline>now)throw cache!.deferred(key);
    _DiskGuideFile? downloaded;
    try {
      await runZoned(()=>providerHttpGet<void>(uri,client:client,operation:HttpOperation(kEpgWorkload),headers:headers,token:token,
        readResponse:(response) async {
          if(response.statusCode!=200)throw StateError('HTTP ${response.statusCode} downloading guide');
          downloaded=response.diskFile;
          if(downloaded==null)throw UnsupportedError('Native guide file was not supplied');
        }),zoneValues:{#tvEpgDiskKey:key});
    } catch(error) {
      if(token?.isCancelled??false)rethrow;
      cache?.failed(key,error);
      if(previous==null)rethrow;
      _diskGuideFiles[destination.path]=previous;return EpgDownloadReceipt();
    }
    _diskGuideFiles[destination.path]=downloaded!;
    if(previous!=null)await previous.close();
    cache?.failed(key);
    return EpgDownloadReceipt(() async {
      if(token?.isCancelled??false)throw const LoadCancelledException();
      await _guideFileRpc(transport,{'action':'commit','token':downloaded!.token});
      downloaded!.committed=true;
      cache?.recordFetchedAt(key,downloaded!.at);
      cache?.failures.remove(key);cache?.failureReasons.remove(key);
      if(cache!=null)cache.revision++;
    });
  }
}

// Browser codec replacement, bounded before inflate writes allocate more data.
class _InflateOutput extends archive.OutputStream {
  final int maximum; _InflateOutput(this.maximum):super(size:32768);
  @override void writeByte(int value) { _checkLimit(length+1,maximum,'decoded gzip'); super.writeByte(value); }
  @override void writeBytes(List<int> bytes,[int? count]) {
    _checkLimit(length+(count??bytes.length),maximum,'decoded gzip'); super.writeBytes(bytes,count);
  }
  @override void writeInputStream(archive.InputStreamBase stream) {
    _checkLimit(length+stream.length,maximum,'decoded gzip'); super.writeInputStream(stream);
  }
}
Uint8List _inflateGzip(List<int> bytes,int maximum) {
  final output=_InflateOutput(maximum);
  archive.GZipDecoder().decodeStream(archive.InputStream(bytes),output);
  final result=Uint8List.fromList(output.getBytes());
  if(bytes.length<8) throw const FormatException('Truncated gzip');
  final footer=ByteData.sublistView(Uint8List.fromList(bytes),bytes.length-8);
  if(footer.getUint32(0,Endian.little)!=archive.getCrc32(result)||footer.getUint32(4,Endian.little)!=result.length) {
    throw const FormatException('Invalid gzip checksum');
  }
  return result;
}

class _WebGzipDecoder extends Converter<List<int>,List<int>> {
  @override List<int> convert(List<int> input)=>_inflateGzip(input,_webEpgMaximumBytes);
  // Retain the 0.1.31 WGT's stream adapter: sending the whole inflated guide as
  // one event makes the XML event parser allocate a whole-guide object graph.
  @override Stream<List<int>> bind(Stream<List<int>> stream) async* {
    if(Zone.current[#tvEpgDisk]==true) {
      final push=_createStreamingGzip.callAsFunction(null,kEpgWorkload.maximumDecodedBytes.toJS) as JSFunction;
      List<Uint8List> decode(List<int> input,bool last) {
        final value=objectJson((push.callAsFunction(null,base64Encode(input).toJS,last.toJS) as JSString).toDart);
        if(value['error']=='limit')throw const HttpWorkloadException('Decoded guide exceeds the work limit',isSizeLimit:true);
        if(value['error']!=null)throw const FormatException('Invalid gzip guide');
        return [for(final chunk in value['chunks'] as List)base64Decode(chunk as String)];
      }
      await for(final bytes in stream) {
        for(var offset=0;offset<bytes.length;offset+=1024) {
          for(final chunk in decode(bytes.sublist(offset,min(offset+1024,bytes.length)),false))yield chunk;
        }
      }
      for(final chunk in decode(const [],true))yield chunk;
      return;

    }

  }

}

Stream<List<XmlEvent>> _xmltvDocumentEvents(Stream<List<XmlEvent>> stream) async* {
  var seen=false,depth=0;
  await for(final events in stream) {
    for(final event in events) {
      if(event is XmlStartElementEvent) {
        if(depth==0) {
          if(seen||event.name!='tv')throw const FormatException('Expected one XMLTV document');
          seen=true;
        }
        if(!event.isSelfClosing)depth++;
      } else if(event is XmlEndElementEvent) {depth--;}
      else if(depth==0&&(event is XmlCDATAEvent||event is XmlTextEvent&&event.value.trim().isNotEmpty)) {
        throw const FormatException('Invalid XMLTV document text');
      }
    }
    yield events;
  }
  if(!seen||depth!=0)throw const FormatException('Incomplete XMLTV document');
}

M3uMediaCatalog _m3uMediaFromJson(Map<String,dynamic> data) {
  if(data['schema']!=1)throw const FormatException('Invalid M3U media cache');
  final categories=<ContentKind,List<MediaCategory>>{}, roots=<ContentKind,List<MediaItem>>{}, children=<String,List<MediaItem>>{};
  for(final entry in (data['categories'] as Map).entries) {
    final kind=ContentKind.values.byName(entry.key as String);
    if(kind!=ContentKind.movie&&kind!=ContentKind.series)throw const FormatException('Invalid M3U root kind');
    categories[kind]=[for(final value in entry.value as List)
      MediaCategory(id:value['id'] as String,title:value['title'] as String,
        kind:ContentKind.values.byName(value['kind'] as String))];
    if(categories[kind]!.any((category)=>category.kind!=kind))throw const FormatException('Invalid M3U category kind');
  }
  for(final entry in (data['roots'] as Map).entries) {
    final kind=ContentKind.values.byName(entry.key as String);
    if(kind!=ContentKind.movie&&kind!=ContentKind.series)throw const FormatException('Invalid M3U root kind');
    roots[kind]=[for(final value in entry.value as List)mediaFromJson(Map<String,dynamic>.from(value as Map))];
    if(roots[kind]!.any((item)=>item.kind!=kind||item.parentId!=null))throw const FormatException('Invalid M3U root item');
  }
  final seriesIds={for(final item in roots[ContentKind.series]??const <MediaItem>[])item.id};
  final seasonIds=<String>{};
  for(final entry in (data['children'] as Map).entries) {
    final parentId=entry.key as String;
    final rows=[for(final value in entry.value as List)mediaFromJson(Map<String,dynamic>.from(value as Map))];
    if(rows.any((item)=>item.parentId!=parentId||
        (item.kind!=ContentKind.season&&item.kind!=ContentKind.episode)))throw const FormatException('Invalid M3U children');
    if(rows.any((item)=>item.kind==ContentKind.season)&&!seriesIds.contains(parentId))throw const FormatException('Invalid M3U series parent');
    for(final item in rows)if(item.kind==ContentKind.season)seasonIds.add(item.id);
    children[parentId]=rows;
  }
  for(final entry in children.entries) {
    if(entry.value.any((item)=>item.kind==ContentKind.episode)&&!seasonIds.contains(entry.key))throw const FormatException('Invalid M3U season parent');
  }
  return M3uMediaCatalog(categories:categories,roots:roots,children:children);
}

String _storeCatalogPart(String raw) {
  final bytes=utf8.encode(raw);
  final compressed=archive.GZipEncoder().encode(bytes,level:6);
  if(compressed==null||compressed.length>=bytes.length)return raw;
  final wrapped=jsonEncode({'format':'gzip-json','decodedBytes':bytes.length,'data':base64Encode(compressed)});
  return utf8.encode(wrapped).length<bytes.length?wrapped:raw;
}
String _catalogPartPayload(String raw,int expectedBytes) {
  final envelope=objectJson(raw);
  if(envelope['format']!='gzip-json') {
    if(envelope.containsKey('format')||utf8.encode(raw).length!=expectedBytes)throw const FormatException('Invalid catalog payload size');
    return raw;
  }
  if(envelope.length!=3||envelope['decodedBytes']!=expectedBytes||expectedBytes<1||expectedBytes>_maximumCatalogPartBytes||
    envelope['data'] is! String)throw const FormatException('Invalid compressed catalog part');
  final encoded=envelope['data'] as String;
  if(encoded.isEmpty||encoded.length>4*((_maximumCatalogPartBytes+2)~/3))throw const FormatException('Compressed catalog part is too large');
  final compressed=base64Decode(encoded);
  if(compressed.isEmpty||compressed.length>_maximumCatalogPartBytes)throw const FormatException('Compressed catalog part is too large');
  final decoded=_inflateGzip(compressed,_maximumCatalogPartBytes);
  if(decoded.length!=expectedBytes)throw const FormatException('Compressed catalog length mismatch');
  return utf8.decode(decoded);
}

  ({List<MediaItem> items,bool known,bool complete,Set<int> pages,Map<String,int> itemPages}) _seriesList(ContentKind kind,String parentId) {
    final items=<String,MediaItem>{},pages=<int>{},itemPages=<String,int>{};
    var known=false,complete=false,total=0;
    for(final saved in _reads.values.toList()) {
      final parent=saved.args['parent'];
      if(saved.args['kind']!=kind.name||parent is! Map||parent['id']!=parentId||
          !const {'mediaItems','allMediaItems','mediaItemsPage'}.contains(saved.method))continue;
      final value=jsonDecode(saved.raw);
      final rows=value is Map?value['items']:value;
      if(rows is! List)continue;

      final page=saved.method=='mediaItemsPage'?value['page'] as int:null;
      if(saved.method=='mediaItemsPage') {
        pages.add(page!);total=value['totalPages'] as int;
      } else if(saved.args['maxPages']==null) { complete=true; }
      for(final row in rows) {

    }

  Future<Map<String,dynamic>> seriesProgress(MediaItem series,List<dynamic> rows,{bool prepareNext=true}) async {
    if(series.kind!=ContentKind.series)throw const FormatException('A series is required');
    final expected=generation,positions=_seriesPositionsFromJson(rows);
    final seasons=_seriesList(ContentKind.season,series.id);
    final seasonIds=seasons.items.map((item)=>item.id).toSet();
    final stored=<String,MediaItem>{};
    for(final raw in rows) {
      if(raw is! Map||raw['item'] is! Map)continue;
      final item=mediaFromJson(Map<String,dynamic>.from(raw['item'] as Map));
      if(item.kind==ContentKind.episode&&seasonIds.contains(item.parentId)&&positions.containsKey(item.id))stored[item.id]=item;
    }
    SeriesWatchProgress progress() {
      final episodes=<String,MediaItem>{};
      for(final season in seasons.items) {
        final list=_seriesList(ContentKind.episode,season.id),ids=list.items.map((item)=>item.id).toSet();
        for(final item in stored.values) {
          if(item.parentId==season.id&&(!list.complete||ids.contains(item.id)))episodes[item.id]=item;
        }
        for(final item in list.items)episodes[item.id]=item;
      }
      return SeriesWatchProgress.fromEpisodes(episodes.values,positions);
    }
    Future<void> prepare(MediaItem season) async {
      if(_seriesList(ContentKind.episode,season.id).known)return;
      await source.connect();
      if(expected!=generation)throw const LoadCancelledException();
      final parent=await mediaParent(season);
      if(parent==null)return;
      final args=<String,dynamic>{'kind':'episode','parent':sourceJson(parent),'page':1};
      final page=await source.mediaItemsPage(ContentKind.episode,parent:parent);
      if(expected!=generation)throw const LoadCancelledException();
      remember('mediaItemsPage',args,jsonEncode(sourceJson(page)));
    }
    var current=progress();
    final latest=current.latestEpisode;
    if(prepareNext&&latest!=null&&current.latestPosition!.isFinished) {
      final index=seasons.items.indexWhere((season)=>season.id==latest.parentId);
      if(index>=0) {
        try {
          await prepare(seasons.items[index]);
          final list=_seriesList(ContentKind.episode,seasons.items[index].id);
          final inSeason=SeriesWatchProgress.fromEpisodes(list.items,positions);
          if(list.complete&&inSeason.nextEpisode==null&&seasons.complete&&index+1<seasons.items.length) {
            await prepare(seasons.items[index+1]);
          }
        } on LoadCancelledException { rethrow; }
        catch(_) { /* Saved history remains available when catalog preparation fails. */ }
        current=progress();
      }
    }
    var next=current.nextEpisode;
    final last=current.latestEpisode;
    if(last!=null&&next!=null&&next.parentId==last.parentId&&next.id!=last.id) {
      final list=_seriesList(ContentKind.episode,last.parentId!);
      if(!list.complete) {
        final from=list.itemPages[last.id],to=list.itemPages[next.id];
        var contiguous=from!=null&&to!=null&&to>=from;
        if(contiguous)for(var page=from!;page<=to!;page++) {
          if(!list.pages.contains(page)) { contiguous=false;break; }
        }
        if(!contiguous)next=null;
      }
    }
    if(last!=null&&next!=null&&next.parentId!=last.parentId) {
      final from=seasons.items.indexWhere((item)=>item.id==last.parentId);
      final to=seasons.items.indexWhere((item)=>item.id==next!.parentId);
      if(!seasons.complete||from<0||to<=from||seasons.items.sublist(from,to)
          .any((season)=>!_seriesList(ContentKind.episode,season.id).complete))next=null;
    }
    final complete=seasons.complete&&seasons.items.isNotEmpty&&current.episodes.isNotEmpty&&
      seasons.items.every((season)=>_seriesList(ContentKind.episode,season.id).complete)&&
      current.watchedCount==current.episodes.length;
    if(expected!=generation)throw const LoadCancelledException();
    return {..._seriesProgressJson(current),'nextEpisode':sourceJson(next),'complete':complete};
  }
  void remember(String method,Map<String,dynamic> args,String raw) {
    if(!_methods.contains(method))return;
    final key=_readKey(method,args), root=_liveRoot(method,args);
    final size=key.length+raw.length;
    if(!root&&size>_maximumOptionalChars)return;
    final previous=_reads.remove(key);
    if(previous!=null&&!root)_optionalChars-=key.length+previous.raw.length;
    while(_reads.length>=_maximumReads||(!root&&_optionalChars+size>_maximumOptionalChars)) {
      final candidates=_reads.entries.where((entry)=>!_liveRoot(entry.value.method,entry.value.args));
      if(candidates.isEmpty)break;
      final oldest=candidates.first;
      _optionalChars-=oldest.key.length+oldest.value.raw.length; _reads.remove(oldest.key);
    }
    final at=DateTime.now().millisecondsSinceEpoch;
    _reads[key]=(method:method,args:args,raw:raw,at:at);
    if(!root)_optionalChars+=size;
    if(_completeAt==null&&read('categories',{})!=null&&read('channels',{})!=null)_completeAt=at;
  }

  Future<void> _prepareSource(String refreshKey,{required bool refresh}) {
    final id=jsonEncode([refreshKey,refresh]), existing=_sourcePreparations[id];
    if(existing!=null)return existing;
    final previous=_sourcePreparationTail;
    final next=previous.catchError((Object _) {}).then((_) async {
      if(source is SharedCatalogSource) {
        await (source as SharedCatalogSource).prepareCatalog(refreshKey,refresh:refresh);
      } else {
        if(refresh&&source is RefreshableSource)(source as RefreshableSource).invalidate();
        await source.connect();
      }
    });
    _sourcePreparationTail=next;
    _sourcePreparations[id]=next;
    if(_sourcePreparations.length>64) {
      final oldest=_sourcePreparations.keys.first;
      if(oldest!=id)_sourcePreparations.remove(oldest);
    }
    return next;
  }
  Future<Map<ContentKind,Object>> _prepareForced(String refreshKey) {
    final existing=_forcedPreparations[refreshKey];
    if(existing!=null)return existing;
    final previous=_forcedPreparationTail;
    final next=previous.catchError((Object _) {}).then((_) async {
      try { await _sourcePreparationTail; } catch(_) {}
      await Future<void>.delayed(Duration.zero);
      await Future.wait(_kindPreparations.values.map((pending)=>pending.then<void>((_) {},onError:(Object _) {})));
      generation++;
      final expected=generation, failures=<ContentKind,Object>{};
      try {
        await _prepareSource(refreshKey,refresh:true);

          try {
            final prepared=await _fetchCatalog(kind);
            if(expected!=generation)throw const LoadCancelledException();
            _install(prepared);
          } catch(error) { failures[kind]=error; }
        }));
      } catch(error) { for(final kind in _catalogKinds)failures[kind]=error; }
      return failures;
    });
    _forcedPreparationTail=next.then<void>((_) {},onError:(Object _) {});
    _forcedPreparations[refreshKey]=next;
    if(_forcedPreparations.length>16) {
      final oldest=_forcedPreparations.keys.first;
      if(oldest!=refreshKey)_forcedPreparations.remove(oldest);
    }
    return next;
  }
  Future<_PreparedCatalog> _fetchCatalog(ContentKind kind) async {
    final at=DateTime.now().millisecondsSinceEpoch;
    if(kind==ContentKind.live) {
      final categories=await source.categories(), channels=await source.channels();
      return _PreparedCatalog(kind:kind,categories:categories,channels:channels,
        pageSize:100,syncedAt:at);
    }
    final categories=await source.mediaCategories(kind);
    final items=<MediaItem>[], seen=<String>{};
    var observedPageSize=0;
    if(source is FullMediaCatalogSource) {
      final fetched=await (source as FullMediaCatalogSource).allMediaItems(kind);
      for(final item in fetched)if(item.id.isNotEmpty&&seen.add(item.id))items.add(item);
    } else {
      final categoryIds=categories.isEmpty?<String?>[null]:categories.map((row)=>row.id).toSet().toList();
      for(final categoryId in categoryIds) {
        for(var page=1;;page++) {
          final fetched=await source.mediaItemsPage(kind,categoryId:categoryId,page:page);
          if(fetched.page!=page)throw StateError('Provider returned an invalid catalog page');
          observedPageSize=math.max(observedPageSize,fetched.items.length);

          if(items.length>_maximumCatalogRows)throw StateError('Provider catalog is too large');
          if(!fetched.hasMore)break;
          if(page>=_maximumCatalogRows||fetched.items.isEmpty)throw StateError('Provider catalog paging did not advance');
        }
      }
    }
    if(items.length>_maximumCatalogRows)throw StateError('Provider catalog is too large');
    final pageSize=source is M3uSource?100:source is XtreamSource?XtreamSource._mediaPageSize:
      math.max(1,observedPageSize);
    final children=kind==ContentKind.series&&source is M3uSource
      ?Map<String,List<MediaItem>>.from((source as M3uSource)._media?.children??const {}):const <String,List<MediaItem>>{};
    return _PreparedCatalog(kind:kind,mediaCategories:categories,items:items,
      children:children,pageSize:pageSize,syncedAt:at);
  }

  Future<Map<String,dynamic>> prepareCatalog(ContentKind kind,{required String refreshKey,required bool forceRefresh}) async {
    if(!_catalogKinds.contains(kind))throw const FormatException('Invalid catalog kind');
    if(refreshKey.isEmpty||refreshKey.length>256)throw const FormatException('A catalog refresh key is required');
    final current=_prepared[kind], now=DateTime.now().millisecondsSinceEpoch;
    final stale=current!=null&&now-current.syncedAt>=CanonicalTvCachePolicy.catalogMaxAge.inMilliseconds;
    if(!forceRefresh&&!stale&&current!=null)return {'kind':kind.name,'count':kind==ContentKind.live?current.channels.length:current.items.length,
      'syncedAt':current.syncedAt,'complete':true};
    if(forceRefresh) {
      final failures=await _prepareForced(refreshKey), failure=failures[kind];
      if(failure!=null)throw failure;
    } else {
      await _prepareSource(refreshKey,refresh:stale);
      final currentGeneration=generation;
      final id=jsonEncode([refreshKey,kind.name]);
      final pending=_kindPreparations[id]??= _fetchCatalog(kind).then((prepared) {
        if(currentGeneration!=generation)throw const LoadCancelledException();
        _install(prepared); return prepared;
      });
      void release() { if(identical(_kindPreparations[id],pending))_kindPreparations.remove(id); }
      pending.then<void>((_)=>release(),onError:(Object _)=>release());
      await pending;
    }
    final prepared=_prepared[kind];
    if(prepared==null)throw StateError('Catalog section is not ready');
    return {'kind':kind.name,'count':kind==ContentKind.live?prepared.channels.length:prepared.items.length,
      'syncedAt':prepared.syncedAt,'complete':true};
  }

  String? readPrepared(String method,Map<String,dynamic> args) {
    final kind=ContentKind.values.byName(args['kind'] as String? ?? 'live');
    final rawParent=args['parent'];
    final prepared=source is M3uSource&&rawParent is Map&&
      const {'mediaItems','mediaItemsPage'}.contains(method)&&
      const {'series','season'}.contains(rawParent['kind'])
      ?_prepared[ContentKind.series]:_prepared[kind];
    if(method=='categories'&&_prepared[ContentKind.live]!=null) {
      return jsonEncode(sourceJson(_prepared[ContentKind.live]!.categories));
    }
    if(method=='channels'&&_prepared[ContentKind.live]!=null) {
      final rows=_prepared[ContentKind.live]!.channels, categoryId=args['categoryId'] as String?;
      return jsonEncode(sourceJson(categoryId==null?rows:rows.where((row)=>row.categoryId==categoryId).toList()));
    }
    if(prepared==null)return null;
    List<MediaItem>? childItems(Map<String,dynamic> raw) {
      final parent=raw['parent'];
      if(parent is! Map||source is! M3uSource)return null;
      final item=mediaFromJson(Map<String,dynamic>.from(parent));
      if(item.kind!=ContentKind.series&&item.kind!=ContentKind.season)return null;
      final series=_prepared[ContentKind.series];
      if(series==null)return null;
      return series.children[item.id]??const <MediaItem>[];
    }
    List<MediaItem> roots({String? categoryId})=>prepared.items.where((item)=>categoryId==null||item.categoryId==categoryId).toList();
    switch(method) {
      case 'mediaCategories':return jsonEncode(sourceJson(prepared.mediaCategories));
      case 'allMediaItems':return jsonEncode(sourceJson(prepared.items));
      case 'mediaItems':
        final children=childItems(args);
        if(args['parent']!=null) return children==null?null:jsonEncode(sourceJson(children));
        if(kind!=ContentKind.movie&&kind!=ContentKind.series)return null;
        var items=roots(categoryId:args['categoryId'] as String?);
        if(source is XtreamSource)items=items.take(prepared.pageSize*(args['maxPages'] as int? ?? 1)).toList();
        return jsonEncode(sourceJson(items));
      case 'mediaItemsPage':
        final children=childItems(args);
        if(args['parent']!=null&&children==null)return null;
        if(args['parent']==null&&kind!=ContentKind.movie&&kind!=ContentKind.series)return null;
        final items=children??roots(categoryId:args['categoryId'] as String?);
        final page=args['page'] as int? ?? 1;
        if(page<1)return null;
        final total=math.max(1,(items.length+prepared.pageSize-1)~/prepared.pageSize).toInt();
        final start=(page-1)*prepared.pageSize;
        return jsonEncode(sourceJson(MediaPage(items:start>=items.length?const []:items.skip(start).take(prepared.pageSize).toList(),
          page:page,totalPages:math.max(page,total).toInt())));
      case 'searchMedia':
        final query=(args['query'] as String? ?? '').trim().toLowerCase();
        if(query.isEmpty)return '[]';
        var matches=roots(categoryId:args['categoryId'] as String?).where((item)=>item.title.toLowerCase().contains(query)).toList();
        if(source is XtreamSource)matches=matches.take(XtreamSource._searchResultLimit).toList();
        return jsonEncode(sourceJson(matches));
      case 'mediaDetails':
        return null;
      default:return null;
    }
  }

  ({List<Map<String,dynamic>> records,int bytes}) _mediaReadChunks(ContentKind kind,{required int byteBudget}) {
    final records=<Map<String,dynamic>>[];

    for(final entry in _reads.entries) {
      final saved=entry.value;
      if(_mediaReadKind(saved.method,saved.args)!=kind||_liveRoot(saved.method,saved.args)||
        entry.key.length+saved.raw.length>_maximumOptionalChars)continue;
      final payload=jsonEncode({'method':saved.method,'args':saved.args,'at':saved.at,'value':jsonDecode(saved.raw)});
      final bytes=utf8.encode(payload);
      if(bytes.isEmpty||bytes.length>2*_maximumOptionalChars||bytes.length>byteBudget-usedBytes)continue;
      final count=(bytes.length+_catalogReadChunkBytes-1)~/_catalogReadChunkBytes;
      final id=_catalogDigest(entry.key);
      for(var index=0;index<count;index++) {
        final start=index*_catalogReadChunkBytes,end=math.min(bytes.length,start+_catalogReadChunkBytes);
        records.add({'type':'mediaReadChunk','id':id,'index':index,'count':count,
          'data':base64Encode(bytes.sublist(start,end))});
      }
      usedBytes+=bytes.length;
    }
    return (records:records,bytes:usedBytes);
  }

        for(final kind in [ContentKind.movie,ContentKind.series]) {
          _prepared[kind]=_PreparedCatalog(kind:kind,mediaCategories:m3uMedia.categories[kind]??const [],
            items:m3uMedia.roots[kind]??const [],children:kind==ContentKind.series?m3uMedia.children:const {},
            pageSize:100,syncedAt:at);
        }
      } else {
        for(final kind in [ContentKind.movie,ContentKind.series]) {
          final raw=restored[_readKey('allMediaItems',{'kind':kind.name})];
          if(raw==null||raw.at>at)continue;
          final categoriesRaw=restored[_readKey('mediaCategories',{'kind':kind.name})];
          final mediaCategories=categoriesRaw==null?const <MediaCategory>[]:
            _decodeRead('mediaCategories',{'kind':kind.name},jsonDecode(categoriesRaw.raw)) as List<MediaCategory>;
          final items=[for(final value in jsonDecode(raw.raw) as List)mediaFromJson(Map<String,dynamic>.from(value as Map))];
          _prepared[kind]=_PreparedCatalog(kind:kind,mediaCategories:mediaCategories,items:items,
            pageSize:source is XtreamSource?XtreamSource._mediaPageSize:14,syncedAt:raw.at);
        }
      }
      final age=Duration(milliseconds:now-at), remaining=CanonicalTvCachePolicy.catalogMaxAge-age;
      return {...(sourceJson(snapshot) as Map<String,dynamic>),'capabilities':_sourceCapabilitiesJson(source),
        'refreshDue':remaining<=Duration.zero,'refreshDelayMs':remaining>Duration.zero?remaining.inMilliseconds:0};
    } catch(_) { return null; }
  }
}

Future<List<Programme>> _sourceGuide(Source source,Map<String,dynamic> args) async {
  final channels=[for(final channel in args['channels'] as List? ?? const [])channelFromJson(Map<String,dynamic>.from(channel))];
  final from=DateTime.tryParse(args['from'] as String? ?? '');
  final to=DateTime.tryParse(args['to'] as String? ?? '');
  final out=<Programme>[];
  void append(List<Programme> batch) {
    for(final programme in batch) {
      if((from==null||programme.stop.isAfter(from))&&(to==null||programme.start.isBefore(to))) out.add(programme);
    }
  }
  if(source is XtreamSource) {
    if(Zone.current[#tvGuideCacheOnly]==true) {
      try { append(await source.epg(channels)); }
      on LoadCancelledException {
        for(final channel in channels) {
          final saved=source._shortEpgCache[channel.id];
          if(saved!=null)append(saved.programmes);
        }
      }
    } else { append(await source.epgForWindow(channels)); }
    return out;
  }
  final batches=source is BatchedEpgSource?(source as BatchedEpgSource).epgBatched(channels):null;
  if(batches==null) { append(await source.epg(channels)); }
  else { await for(final batch in batches) { append(batch); } }
  return out;
}
// EPG failures cross the Worker boundary as a reserved DTO. Exception text can
// contain provider URLs or XML content, so only fixed categories and HTTP codes

const _webGuideCauseCodes={'epg_timeout','epg_network','epg_size_limit',
  'epg_invalid_data','epg_http','epg_unsupported','epg_unavailable'};
class _WebGuideDeferred extends StateError {
  final int retryAt;
  final Map<String,Object> cause;
  _WebGuideDeferred(this.retryAt,this.cause):super(cause['code']=='epg_http'
    ?'HTTP ${cause['status']} downloading guide':'Guide retry deferred');
  @override String toString()=>'Guide retry deferred';
}
Map<String,Object>? _webGuideReasonFromJson(Object? raw) {
  if(raw is! Map||raw.keys.any((key)=>key!='code'&&key!='status'))return null;
  final code=raw['code'];
  if(code is! String||!_webGuideCauseCodes.contains(code))return null;
  final status=raw['status'];
  if(status!=null&&(code!='epg_http'||status is! int||status<100||status>599))return null;
  return {'code':code,if(status!=null)'status':status as int};
}
Map<String,Object> _webGuideCause(Object error) {
  if(error is _WebGuideDeferred)return error.cause;
  final failure=_webGuideFailure(error);
  return {'code':failure['code']!,if(failure['status']!=null)'status':failure['status']!};
}
Map<String,Object> _webGuideFailure(Object error) {
  if(error is _WebGuideDeferred)return {'__tvbroEpgError':1,'code':'epg_retry_deferred',
    'causeCode':error.cause['code']!,'retryAt':error.retryAt,
    if(error.cause['status']!=null)'status':error.cause['status']!};
  var code='epg_unavailable';
  int? status;
  if(error is TimeoutException)code='epg_timeout';
  else if(error is HttpWorkloadException)code=error.isSizeLimit?'epg_size_limit':'epg_invalid_data';
  else if(error is FormatException||error is XmltvParseException)code='epg_invalid_data';
  else if(error is SocketException)code='epg_network';
  else if(error is ProviderRequestBlocked) { code='epg_http';status=error.status; }
  else if(error is UnsupportedError)code='epg_unsupported';
  else if(error is StateError) {
    if(error.message=='Guide retry deferred')code='epg_retry_deferred';
    final http=RegExp(r'^HTTP ([1-5][0-9]{2}) (?:downloading guide$|from )').firstMatch(error.message);
    if(http!=null) { code='epg_http';status=int.parse(http.group(1)!); }
  }
  return {'__tvbroEpgError':1,'code':code,
    if(status!=null&&status>=100&&status<=599)'status':status};
}
